AI在保险行业的应用场景全景与Java技术实现映射

一、场景全景概览

保险行业是AI应用最为密集的金融领域之一。从客户获取、核保风控到理赔服务，AI几乎可以渗透到保险价值链的每一个环节。以下是七个核心应用场景的系统化分析，每个场景都有完整的Java技术实现方案。

二、智能核保（Intelligent Underwriting）

2.1 场景描述

智能核保是AI在保险行业最成熟的应用之一。传统核保依赖人工审核大量健康告知问卷和体检报告，效率低、主观性强。AI驱动的智能核保系统能够：

  • 自动解析健康告知中的自由文本和结构化数据
  • 结合规则引擎和ML模型进行风险评分
  • 实现"自动通过 / 加费承保 / 除外承保 / 拒保 / 转人工"的五级分流

2.2 业务流程

投保申请 → 健康告知智能解析 → 规则引擎初筛 → ML风险评分 → 核保决策 → 出单/加费/拒保/转人工
              ↑                    ↑               ↑
         (NLP提取症状/疾病)    (Drools规则)    (XGBoost模型)

2.3 Java实现

java
// UnderwritingService.java — 智能核保服务
package com.insurance.ai.underwriting;

import com.insurance.domain.underwriting.UnderwritingRequest;
import com.insurance.domain.underwriting.UnderwritingResult;
import com.insurance.domain.underwriting.HealthDeclaration;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.ai.tool.annotation.Tool;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Map;

@Service
public class UnderwritingService {

    private final ChatClient chatClient;
    private final UnderwritingRuleEngine ruleEngine;
    private final RiskScoringModel riskModel;

    public UnderwritingService(ChatModel chatModel,
                               UnderwritingRuleEngine ruleEngine,
                               RiskScoringModel riskModel) {
        this.chatClient = ChatClient.builder(chatModel).build();
        this.ruleEngine = ruleEngine;
        this.riskModel = riskModel;
    }

    /**
     * 智能核保主流程
     * @param request 核保请求（包含健康告知文本和结构化数据）
     * @return 核保结果
     */
    public UnderwritingResult underwrite(UnderwritingRequest request) {
        // Step 1: 健康告知智能解析 — 使用LLM提取关键医学实体
        var healthEntities = parseHealthDeclaration(request.healthDeclaration());

        // Step 2: 规则引擎初筛 — 基于监管和公司核保规则
        var ruleResult = ruleEngine.evaluate(request, healthEntities);

        // 如果规则引擎直接决定了结果（如明确拒保），直接返回
        if (ruleResult.isDecisive()) {
            return ruleResult.toUnderwritingResult();
        }

        // Step 3: ML模型风险评分 — 对不确定的案件进行量化评估
        var riskScore = riskModel.score(healthEntities, request.insurantProfile());

        // Step 4: 综合决策
        return makeDecision(riskScore, ruleResult);
    }

    /**
     * 使用LLM解析健康告知文本，提取结构化医疗实体
     */
    public List<HealthEntity> parseHealthDeclaration(HealthDeclaration declaration) {
        var systemPrompt = """
            你是一个医疗保险核保专家。请从以下健康告知文本中提取关键医学实体。
            对于每个提取的实体，请标注：
            - entity_type: DISEASE / SYMPTOM / SURGERY / MEDICATION / LAB_RESULT
            - severity: MILD / MODERATE / SEVERE
            - icd10_code: 对应的ICD-10编码（如能确定）
            - is_critical: 是否属于保险公司重点关注的高风险疾病

            以JSON数组格式输出。
            """;

        var response = chatClient.prompt()
                .system(systemPrompt)
                .user(declaration.freeText())
                .call()
                .content();

        return parseHealthEntitiesFromJson(response);
    }

    private UnderwritingResult makeDecision(double riskScore,
                                             RuleEngineResult ruleResult) {
        if (riskScore < 0.3) {
            return UnderwritingResult.standardApproval();
        } else if (riskScore < 0.6) {
            return UnderwritingResult.approvedWithLoading(
                calculateLoading(riskScore)
            );
        } else if (riskScore < 0.8) {
            return UnderwritingResult.approvedWithExclusion(
                ruleResult.highRiskConditions()
            );
        } else {
            return UnderwritingResult.referToManual(riskScore, ruleResult);
        }
    }

    private double calculateLoading(double riskScore) {
        // 加费比例 = 基础费率 × (1 + 风险系数 × 调整因子)
        return 1.0 + riskScore * 0.5;  // 最高加费50%
    }

    private List<HealthEntity> parseHealthEntitiesFromJson(String json) {
        // 使用 Jackson 解析 LLM 返回的 JSON
        try {
            var mapper = new com.fasterxml.jackson.databind.ObjectMapper();
            return mapper.readValue(json,
                new com.fasterxml.jackson.core.type.TypeReference<List<HealthEntity>>() {});
        } catch (Exception e) {
            throw new RuntimeException("Failed to parse health entities from LLM response", e);
        }
    }
}

// UnderwritingResult.java
record UnderwritingResult(
    UnderwritingDecision decision,
    double riskScore,
    double loadingRatio,       // 加费比例（仅在加费承保时有效）
    List<String> exclusions,   // 除外责任（仅在除外承保时有效）
    String reason,
    boolean needManualReview
) {
    static UnderwritingResult standardApproval() {
        return new UnderwritingResult(
            UnderwritingDecision.APPROVED, 0.2, 1.0,
            List.of(), "标准体承保", false
        );
    }

    static UnderwritingResult approvedWithLoading(double loading) {
        return new UnderwritingResult(
            UnderwritingDecision.APPROVED_WITH_LOADING, 0.5, loading,
            List.of(), "次标准体，加费承保", false
        );
    }

    static UnderwritingResult approvedWithExclusion(List<String> exclusions) {
        return new UnderwritingResult(
            UnderwritingDecision.APPROVED_WITH_EXCLUSION, 0.7, 1.0,
            exclusions, "除外承保", false
        );
    }

    static UnderwritingResult referToManual(double score, RuleEngineResult result) {
        return new UnderwritingResult(
            UnderwritingDecision.REFER_TO_MANUAL, score, 1.0,
            List.of(), "高风险案件，转人工核保", true
        );
    }
}

enum UnderwritingDecision {
    APPROVED, APPROVED_WITH_LOADING, APPROVED_WITH_EXCLUSION,
    DECLINED, REFER_TO_MANUAL
}

2.4 规则引擎实现（Drools）

java
// UnderwritingRuleEngine.java — 基于Drools的核保规则引擎
package com.insurance.ai.underwriting;

import org.kie.api.KieServices;
import org.kie.api.runtime.KieContainer;
import org.kie.api.runtime.KieSession;
import org.springframework.stereotype.Component;

import java.util.List;

@Component
public class UnderwritingRuleEngine {

    private final KieContainer kieContainer;

    public UnderwritingRuleEngine() {
        var kieServices = KieServices.Factory.get();
        this.kieContainer = kieServices.getKieClasspathContainer();
    }

    public RuleEngineResult evaluate(UnderwritingRequest request,
                                      List<HealthEntity> healthEntities) {
        var kieSession = kieContainer.newKieSession();
        var result = new RuleEngineResult();

        try {
            kieSession.insert(request);
            healthEntities.forEach(kieSession::insert);
            kieSession.insert(result);
            kieSession.fireAllRules();
        } finally {
            kieSession.dispose();
        }

        return result;
    }
}

// RuleEngineResult.java
class RuleEngineResult {
    private boolean decisive = false;
    private UnderwritingDecision decision;
    private List<String> highRiskConditions = List.of();
    private String reason;

    public void setDecision(UnderwritingDecision decision, String reason) {
        this.decision = decision;
        this.reason = reason;
        this.decisive = true;
    }

    public boolean isDecisive() { return decisive; }
    public List<String> highRiskConditions() { return highRiskConditions; }

    UnderwritingResult toUnderwritingResult() {
        return new UnderwritingResult(
            decision, 0.0, 1.0, List.of(), reason,
            decision == UnderwritingDecision.REFER_TO_MANUAL
        );
    }
}

三、产品推荐与匹配引擎

3.1 场景描述

基于客户画像（年龄、性别、职业、家庭结构、收入水平、已有保障、预算约束）智能推荐最合适的保险产品组合。核心技术栈：规则引擎 + 向量相似度匹配 + 大模型方案解读。

3.2 Java实现

java
// ProductRecommendationService.java
package com.insurance.ai.recommendation;

import com.insurance.domain.product.InsuranceProduct;
import com.insurance.domain.customer.CustomerProfile;
import org.springframework.ai.document.Document;
import org.springframework.ai.embedding.EmbeddingModel;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.util.*;
import java.util.stream.Collectors;

@Service
public class ProductRecommendationService {

    private final ProductRepository productRepository;
    private final EmbeddingModel embeddingModel;
    private final VectorStore vectorStore;
    private final RecommendationRuleEngine ruleEngine;

    public ProductRecommendationService(ProductRepository productRepository,
                                         EmbeddingModel embeddingModel,
                                         VectorStore vectorStore,
                                         RecommendationRuleEngine ruleEngine) {
        this.productRepository = productRepository;
        this.embeddingModel = embeddingModel;
        this.vectorStore = vectorStore;
        this.ruleEngine = ruleEngine;
    }

    /**
     * 混合推荐：规则筛选 + 向量相似度排序
     */
    public RecommendationResult recommend(CustomerProfile profile) {
        // 1. 规则引擎硬筛选：排除不符合投保条件的产品
        var eligibleProducts = ruleEngine.filterEligibleProducts(
            productRepository.findAll(), profile
        );

        if (eligibleProducts.isEmpty()) {
            return RecommendationResult.empty("没有符合条件的产品");
        }

        // 2. 构建客户需求向量（自然语言描述客户画像和需求）
        var customerNeedVector = buildCustomerNeedVector(profile);

        // 3. 向量相似度匹配：对合格产品按需求匹配度排序
        var scoredProducts = eligibleProducts.stream()
            .map(product -> {
                var productVector = getProductEmbedding(product);
                var similarity = cosineSimilarity(customerNeedVector, productVector);
                return new ScoredProduct(product, similarity);
            })
            .sorted(Comparator.comparingDouble(ScoredProduct::score).reversed())
            .limit(5)
            .toList();

        // 4. 使用LLM生成推荐理由和方案对比
        var recommendationText = generateRecommendationText(profile, scoredProducts);

        return new RecommendationResult(scoredProducts, recommendationText);
    }

    private double[] buildCustomerNeedVector(CustomerProfile profile) {
        var profileText = String.format("""
            客户画像：
            - 年龄：%d岁
            - 性别：%s
            - 职业类别：%d级
            - 家庭结构：%s
            - 年收入：%s元
            - 已有保障：%s
            - 预算：每年%s元
            - 偏好险种：%s
            - 关注点：%s
            """,
            profile.age(), profile.gender(), profile.occupationClass(),
            profile.familyStructure(), profile.annualIncome(),
            profile.existingCoverage(), profile.budget(),
            profile.preferredType(), profile.concerns()
        );

        return embeddingModel.embed(profileText);
    }

    private double[] getProductEmbedding(InsuranceProduct product) {
        var productText = String.format("""
            产品名称：%s
            险种类型：%s
            适用年龄：%d-%d岁
            适用职业：1-%d类
            保障期限：%s
            核心保障：%s
            产品特色：%s
            """,
            product.name(), product.type(),
            product.minAge(), product.maxAge(),
            product.maxOccupationClass(),
            product.coveragePeriod(),
            product.coreCoverage(),
            product.features()
        );

        return embeddingModel.embed(productText);
    }

    private double cosineSimilarity(double[] a, double[] b) {
        double dotProduct = 0.0;
        double normA = 0.0;
        double normB = 0.0;
        for (int i = 0; i < a.length; i++) {
            dotProduct += a[i] * b[i];
            normA += a[i] * a[i];
            normB += b[i] * b[i];
        }
        if (normA == 0 || normB == 0) return 0.0;
        return dotProduct / (Math.sqrt(normA) * Math.sqrt(normB));
    }

    private String generateRecommendationText(CustomerProfile profile,
                                               List<ScoredProduct> products) {
        // 使用LLM生成推荐理由（实际项目中调用ChatClient）
        var productList = products.stream()
            .map(sp -> String.format("- %s (匹配度: %.0f%%)",
                sp.product().name(), sp.score() * 100))
            .collect(Collectors.joining("\n"));

        return "基于您的需求，为您推荐以下产品：\n" + productList +
               "\n\n以上推荐综合考虑了您的年龄、职业、预算和保障需求。";
    }
}

// RecommendationRuleEngine.java
package com.insurance.ai.recommendation;

import com.insurance.domain.product.InsuranceProduct;
import org.jeasy.rules.api.Facts;
import org.jeasy.rules.api.Rules;
import org.jeasy.rules.core.DefaultRulesEngine;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;

@Component
public class RecommendationRuleEngine {

    public List<InsuranceProduct> filterEligibleProducts(
            List<InsuranceProduct> allProducts,
            CustomerProfile profile) {

        var eligibleProducts = new ArrayList<InsuranceProduct>();

        for (var product : allProducts) {
            var facts = new Facts();
            facts.put("profile", profile);
            facts.put("product", product);
            facts.put("eligible", true);

            var rules = new Rules();
            rules.register(new AgeRule());
            rules.register(new OccupationClassRule());
            rules.register(new BudgetRule());

            var engine = new DefaultRulesEngine();
            engine.fire(rules, facts);

            if (facts.get("eligible")) {
                eligibleProducts.add(product);
            }
        }

        return eligibleProducts;
    }
}

四、条款智能解读（RAG + QA）

4.1 场景描述

保险条款动辄几万字，专业术语密集，用户很难阅读理解。RAG（检索增强生成）方案可以：

  • 将PDF条款文档解析后存入向量数据库
  • 用户用自然语言提问（"等待期是多少天""高血压能不能投保"）
  • 系统检索相关条款片段，LLM基于原文生成答案并注明出处

4.2 Java实现（Spring AI RAG）

java
// ClauseQAService.java — 保险条款智能问答服务
package com.insurance.ai.clause;

import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.ai.document.Document;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.util.List;
import java.util.stream.Collectors;

@Service
public class ClauseQAService {

    private final VectorStore vectorStore;
    private final ChatClient chatClient;
    private final ClauseDocumentParser documentParser;

    public ClauseQAService(VectorStore vectorStore,
                            ChatModel chatModel,
                            ClauseDocumentParser documentParser) {
        this.vectorStore = vectorStore;
        this.chatClient = ChatClient.builder(chatModel).build();
        this.documentParser = documentParser;
    }

    /**
     * 上传并解析条款文档
     */
    public ClauseUploadResult uploadClause(MultipartFile file, String productCode) {
        // 使用 Apache PDFBox + Tika 解析文档
        var parsedDocs = documentParser.parsePdf(file);

        // 添加元数据
        for (var doc : parsedDocs) {
            doc.getMetadata().put("product_code", productCode);
            doc.getMetadata().put("source_file", file.getOriginalFilename());
            doc.getMetadata().put("upload_time",
                java.time.LocalDateTime.now().toString());
        }

        // 写入向量数据库
        vectorStore.add(parsedDocs);

        return new ClauseUploadResult(
            productCode,
            file.getOriginalFilename(),
            parsedDocs.size(),
            parsedDocs.stream().mapToInt(d -> d.getContent().length()).sum()
        );
    }

    /**
     * RAG问答：用户自然语言提问，检索相关条款，生成答案和原文引用
     */
    public ClauseAnswer answer(String question, String productCode) {
        // 1. 语义检索相关条款片段
        var searchRequest = SearchRequest.query(question)
                .withTopK(5)
                .withSimilarityThreshold(0.7);

        // 如果指定了产品代码，增加过滤条件
        var retrievedDocs = vectorStore.similaritySearch(searchRequest);

        if (retrievedDocs.isEmpty()) {
            return new ClauseAnswer(
                "抱歉，未在条款中找到相关信息。",
                "建议您查看完整条款文件或联系客服。",
                List.of()
            );
        }

        // 2. 构建提示词：包含检索到的条款原文和用户问题
        var context = retrievedDocs.stream()
                .map(doc -> String.format("【条款来源：%s，章节：%s】\n%s",
                    doc.getMetadata().getOrDefault("source_file", "未知"),
                    doc.getMetadata().getOrDefault("chapter", "未知章节"),
                    doc.getContent()))
                .collect(Collectors.joining("\n\n---\n\n"));

        var systemPrompt = """
            你是一个专业的保险条款解读助手。请基于以下条款原文回答用户的问题。

            要求：
            1. 只基于提供的条款原文回答问题，不要添加原文中没有的信息
            2. 回答要通俗易懂，将专业术语解释清楚
            3. 引用原文关键段落，标注来源章节
            4. 如果原文信息不足以回答，请明确告知
            5. 在回答末尾标明引用来源

            条款原文：
            %s
            """.formatted(context);

        // 3. 使用LLM生成回答
        var answer = chatClient.prompt()
                .system(systemPrompt)
                .user(question)
                .call()
                .content();

        // 4. 构建引用列表
        var citations = retrievedDocs.stream()
                .map(doc -> new Citation(
                    doc.getMetadata().getOrDefault("chapter", "未知章节"),
                    doc.getContent().substring(0,
                        Math.min(200, doc.getContent().length())) + "..."
                ))
                .toList();

        return new ClauseAnswer(answer,
            "以上回答基于条款原文，仅供参考。具体以正式保单条款为准。",
            citations
        );
    }
}

import org.springframework.stereotype.Component;
import org.springframework.web.multipart.MultipartFile;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;

@Component
public class ClauseDocumentParser {

    private static final Tika tika = new Tika();
    private static final int CHUNK_SIZE = 1000;  // 每块1000字符
    private static final int CHUNK_OVERLAP = 200; // 块间重叠200字符

    public List<Document> parsePdf(MultipartFile file) {
        var documents = new ArrayList<Document>();

        try (var pdfDoc = Loader.loadPDF(file.getBytes())) {
            var stripper = new PDFTextStripper();
            var fullText = stripper.getText(pdfDoc);

            // 按章节分割
            var chapters = splitByChapters(fullText);

            for (var chapter : chapters) {
                // 对每个章节的内容进行分块
                var chunks = chunkText(chapter.content(), CHUNK_SIZE, CHUNK_OVERLAP);
                for (int i = 0; i < chunks.size(); i++) {
                    var doc = new Document(
                        chunks.get(i),
                        Map.of(
                            "chapter", chapter.title(),
                            "chunk_index", String.valueOf(i),
                            "total_chunks", String.valueOf(chunks.size())
                        )
                    );
                    documents.add(doc);
                }
            }

        } catch (Exception e) {
            throw new RuntimeException("Failed to parse PDF: " + e.getMessage(), e);
        }

        return documents;
    }

    private List<Chapter> splitByChapters(String text) {
        // 按常见条款章节标题进行分割
        var chapters = new ArrayList<Chapter>();
        var lines = text.split("\n");
        var currentTitle = "总则";
        var currentContent = new StringBuilder();

        for (var line : lines) {
            // 检测章节标题（以"第X条"、"第X章"开头或全为大写标题）
            if (line.matches("^第[一二三四五六七八九十百]+[条章].*") ||
                line.matches("^[一二三四五六七八九十]、.*")) {
                if (!currentContent.isEmpty()) {
                    chapters.add(new Chapter(currentTitle, currentContent.toString().trim()));
                }
                currentTitle = line.trim();
                currentContent = new StringBuilder();
            } else {
                currentContent.append(line).append("\n");
            }
        }

        if (!currentContent.isEmpty()) {
            chapters.add(new Chapter(currentTitle, currentContent.toString().trim()));
        }

        return chapters.isEmpty() ?
            List.of(new Chapter("全文", text)) : chapters;
    }

    private List<String> chunkText(String text, int chunkSize, int overlap) {
        var chunks = new ArrayList<String>();
        if (text.length() <= chunkSize) {
            chunks.add(text);
            return chunks;
        }

        int start = 0;
        while (start < text.length()) {
            int end = Math.min(start + chunkSize, text.length());

            // 尽量在句号或换行处断开
            if (end < text.length()) {
                var breakPoint = text.lastIndexOf("。", end);
                if (breakPoint > start + chunkSize / 2) {
                    end = breakPoint + 1;
                }
            }

            chunks.add(text.substring(start, end));
            start = end - overlap;
        }

        return chunks;
    }

    record Chapter(String title, String content) {}
}

package com.insurance.ai.claim;

import com.insurance.domain.claim.Claim;

import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.ArrayList;
import java.util.List;

@Service
public class ClaimProcessingService {

    private final ChatClient chatClient;
    private final ClaimRuleEngine claimRuleEngine;
    private final DocumentRecognitionService docRecognitionService;

    public ClaimProcessingService(ChatModel chatModel,
                                   ClaimRuleEngine claimRuleEngine,
                                   DocumentRecognitionService docRecognitionService) {
        this.chatClient = ChatClient.builder(chatModel).build();
        this.claimRuleEngine = claimRuleEngine;
        this.docRecognitionService = docRecognitionService;
    }

    /**
     * 处理理赔申请
     */
    public ClaimResult processClaim(Claim claim, List<MultipartFile> documents) {
        // 1. 材料识别与分类
        var recognizedDocs = new ArrayList<RecognizedDocument>();
        for (var file : documents) {
            try {
                var recognized = docRecognitionService.recognize(file);
                recognizedDocs.add(recognized);
            } catch (Exception e) {
                return ClaimResult.materialError("材料识别失败: " + file.getOriginalFilename());
            }
        }

        // 2. 结构化信息提取
        var extractedInfo = extractClaimInfo(recognizedDocs, claim.getClaimType().name());

        // 3. 规则引擎判断
        var ruleResult = claimRuleEngine.evaluate(claim, extractedInfo);

        // 4. 计算赔付金额
        var payableAmount = calculatePayableAmount(claim, extractedInfo, ruleResult);

        // 5. 生成理赔决定书
        var decisionLetter = generateDecisionLetter(claim, ruleResult, payableAmount);

        return new ClaimResult(
            ruleResult.decision(),
            payableAmount,
            ruleResult.reasons(),
            decisionLetter,
            ruleResult.needManualReview()
        );
    }

    /**
     * 从识别后的材料中提取结构化理赔信息
     */
    private ExtractedClaimInfo extractClaimInfo(List<RecognizedDocument> docs,
                                                  String claimType) {
        var context = docs.stream()
            .map(d -> String.format("[%s] %s", d.documentType(), d.extractedText()))
            .reduce("", (a, b) -> a + "\n" + b);

        var systemPrompt = """
            从以下理赔材料中提取关键信息，以JSON格式返回：
            {
                "diagnosis": "诊断名称",
                "diagnosisCode": "ICD-10编码（如有）",
                "hospitalName": "就诊医院名称",
                "admissionDate": "入院日期",
                "dischargeDate": "出院日期",
                "totalExpense": 总费用（元），
                "socialInsurancePaid": 社保支付金额，
                "selfPaid": 自付金额，
                "accidentCause": "事故原因（如为意外事故理赔）",
                "accidentDate": "事故发生日期",
                "disabilityLevel": "伤残等级1-10（如适用）",
                "isWithinCoverage": true/false
            }
            """;

        var response = chatClient.prompt()
                .system(systemPrompt)
                .user(context)
                .call()
                .content();

        // 解析JSON
        try {
            var mapper = new com.fasterxml.jackson.databind.ObjectMapper();
            return mapper.readValue(response, ExtractedClaimInfo.class);
        } catch (Exception e) {
            throw new RuntimeException("信息提取失败", e);
        }
    }

    /**
     * 计算赔付金额
     */
    private BigDecimal calculatePayableAmount(Claim claim,
                                               ExtractedClaimInfo info,
                                               RuleEngineResult ruleResult) {
        if (ruleResult.decision() != ClaimDecision.APPROVED &&
            ruleResult.decision() != ClaimDecision.PARTIALLY_APPROVED) {
            return BigDecimal.ZERO;
        }

        var payable = info.selfPaid();

        // 减去免赔额
        if (claim.getDeductible() != null) {
            payable = payable.subtract(claim.getDeductible());
            if (payable.compareTo(BigDecimal.ZERO) < 0) {
                payable = BigDecimal.ZERO;
            }
        }

        // 应用赔付比例
        if (claim.getCoinsuranceRatio() != null) {
            payable = payable.multiply(claim.getCoinsuranceRatio());
        }

        // 不超过保额上限
        var sumInsured = claim.getPolicy().getSumInsured();
        if (payable.compareTo(sumInsured) > 0) {
            payable = sumInsured;
        }

        return payable.setScale(2, RoundingMode.HALF_UP);
    }

    private String generateDecisionLetter(Claim claim, RuleEngineResult result,
                                           BigDecimal amount) {
        return switch (result.decision()) {
            case APPROVED -> String.format(
                "经审核，您的事故属于保险责任范围，核定赔付金额：%s元。", amount);
            case PARTIALLY_APPROVED -> String.format(
                "经审核，部分费用属于保险责任范围，核定赔付金额：%s元。原因：%s",
                amount, String.join("；", result.reasons()));
            case REJECTED -> String.format(
                "经审核，您的事故不属于保险责任范围。原因：%s",
                String.join("；", result.reasons()));
            case PENDING_INFO -> "请补充以下材料：" + String.join("、", result.missingDocuments());
        };
    }
}

// FraudDetectionService.java
package com.insurance.ai.fraud;

import com.insurance.domain.claim.Claim;
import org.springframework.stereotype.Service;

import java.util.*;
import java.util.stream.Collectors;

@Service
public class FraudDetectionService {

    /**
     * 理赔反欺诈检测
     */
    public FraudDetectionResult detect(Claim claim,
                                        List<HistoricalClaim> historicalClaims) {

        var flags = new ArrayList<FraudFlag>();
        double totalScore = 0.0;

        // 1. 短期出险检测（投保后30天内出险）
        if (isShortTermClaim(claim)) {
            flags.add(new FraudFlag("短期出险", 30.0,
                "投保后30天内发生保险事故"));
            totalScore += 30.0;
        }

        // 2. 历史理赔频率异常
        var frequencyScore = checkClaimFrequency(claim, historicalClaims);
        if (frequencyScore > 0) {
            flags.add(new FraudFlag("高频理赔", frequencyScore,
                "近12个月理赔次数异常偏高"));
            totalScore += frequencyScore;
        }

        // 3. 重复理赔检测
        var duplicateScore = checkDuplicateClaims(claim, historicalClaims);
        if (duplicateScore > 0) {
            flags.add(new FraudFlag("疑似重复理赔", duplicateScore,
                "发现相同金额或相同医院的近期理赔记录"));
            totalScore += duplicateScore;
        }

        // 4. 就诊医院风险评级
        var hospitalScore = checkHospitalRisk(claim);
        if (hospitalScore > 0) {
            flags.add(new FraudFlag("高风险医院", hospitalScore,
                "就诊医院位于高风险名单"));
            totalScore += hospitalScore;
        }

        // 5. 理赔时间模式异常
        var timePatternScore = checkTimePattern(claim, historicalClaims);
        if (timePatternScore > 0) {
            flags.add(new FraudFlag("时间模式异常", timePatternScore,
                "出险时间模式不符合常见规律"));
            totalScore += timePatternScore;
        }

        // 综合判断
        var riskLevel = categorizeRisk(totalScore);
        var needManualReview = riskLevel == FraudRiskLevel.HIGH ||
                                riskLevel == FraudRiskLevel.CRITICAL;

        return new FraudDetectionResult(
            riskLevel, totalScore, flags, needManualReview,
            generateRecommendation(riskLevel, flags)
        );
    }

    private boolean isShortTermClaim(Claim claim) {
        var effectiveDate = claim.getPolicy().getEffectiveDate();
        var incidentDate = claim.getIncidentDate();
        if (effectiveDate == null || incidentDate == null) return false;
        return java.time.temporal.ChronoUnit.DAYS.between(effectiveDate, incidentDate) <= 30;
    }

    private double checkClaimFrequency(Claim claim,
                                        List<HistoricalClaim> historicalClaims) {
        var oneYearAgo = java.time.LocalDate.now().minusYears(1);
        var recentClaims = historicalClaims.stream()
            .filter(h -> h.claimDate().isAfter(oneYearAgo))
            .count();

        if (recentClaims >= 5) return 25.0;
        if (recentClaims >= 3) return 15.0;
        if (recentClaims >= 2) return 5.0;
        return 0.0;
    }

    private double checkDuplicateClaims(Claim claim,
                                          List<HistoricalClaim> historicalClaims) {
        var claimAmount = claim.getClaimedAmount();
        if (claimAmount == null) return 0.0;

        var duplicates = historicalClaims.stream()
            .filter(h -> h.claimedAmount() != null)
            .filter(h -> h.claimedAmount().compareTo(claimAmount) == 0)
            .count();

        if (duplicates >= 2) return 20.0;
        if (duplicates >= 1) return 10.0;
        return 0.0;
    }

    private double checkHospitalRisk(Claim claim) {
        // 实际实现应查询高风险医院名单数据库
        return 0.0;
    }

    private double checkTimePattern(Claim claim,
                                     List<HistoricalClaim> historicalClaims) {
        // 检查是否所有理赔都集中在周末或节假日
        var weekendClaims = historicalClaims.stream()
            .filter(h -> {
                var dow = h.incidentDate().getDayOfWeek();
                return dow == java.time.DayOfWeek.SATURDAY ||
                       dow == java.time.DayOfWeek.SUNDAY;
            })
            .count();

        var totalClaims = historicalClaims.size();
        if (totalClaims >= 3 && (double) weekendClaims / totalClaims > 0.8) {
            return 15.0;
        }
        return 0.0;
    }

    private FraudRiskLevel categorizeRisk(double totalScore) {
        if (totalScore >= 70) return FraudRiskLevel.CRITICAL;
        if (totalScore >= 50) return FraudRiskLevel.HIGH;
        if (totalScore >= 30) return FraudRiskLevel.MEDIUM;
        return FraudRiskLevel.LOW;
    }

    private String generateRecommendation(FraudRiskLevel level,
                                           List<FraudFlag> flags) {
        return switch (level) {
            case CRITICAL -> "高度疑似欺诈，建议立即启动调查程序，暂缓赔付。风险点：" +
                flags.stream().map(FraudFlag::description)
                    .collect(Collectors.joining("；"));
            case HIGH -> "存在较多风险信号，建议人工复核后决定是否赔付。";
            case MEDIUM -> "存在少量风险信号，可正常审核流程但建议关注。";
            case LOW -> "未发现明显风险信号，可正常处理。";
        };
    }
}

// AgentConversationManager.java — 对话管理与上下文保持
package com.insurance.ai.agent;

import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

@Component
public class AgentConversationManager {

    // 使用 ConcurrentHashMap 存储会话历史
    private final Map<String, List<ConversationTurn>> sessions = new ConcurrentHashMap<>();
    private static final int MAX_HISTORY = 20;  // 最多保留20轮对话

    public void addTurn(String sessionId, String userMessage, String agentResponse) {
        sessions.computeIfAbsent(sessionId, k -> new ArrayList<>())
                .add(new ConversationTurn(userMessage, agentResponse));

        // 保持历史在合理范围内
        var history = sessions.get(sessionId);
        while (history.size() > MAX_HISTORY) {
            history.removeFirst();
        }
    }

    public List<ConversationTurn> getHistory(String sessionId) {
        return sessions.getOrDefault(sessionId, List.of());
    }

    public void clearSession(String sessionId) {
        sessions.remove(sessionId);
    }

    /**
     * 构建用于LLM的对话历史文本
     */
    public String buildHistoryContext(String sessionId) {
        var history = getHistory(sessionId);
        if (history.isEmpty()) return "";

        var sb = new StringBuilder("对话历史：\n");
        for (var turn : history) {
            sb.append("用户：").append(turn.userMessage()).append("\n");
            sb.append("助手：").append(turn.agentResponse()).append("\n");
        }
        return sb.toString();
    }

    record ConversationTurn(String userMessage, String agentResponse) {}
}

// PremiumQuoteService.java — 报价与保费测算
package com.insurance.ai.quote;

import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.model.ChatModel;
import org.springframework.ai.tool.annotation.Tool;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

@Service
public class PremiumQuoteService {

    private final ChatClient chatClient;
    private final PremiumCalculator premiumCalculator;
    private final ProductRepository productRepository;

    public PremiumQuoteService(ChatModel chatModel,
                                PremiumCalculator premiumCalculator,
                                ProductRepository productRepository) {
        this.chatClient = ChatClient.builder(chatModel).build();
        this.premiumCalculator = premiumCalculator;
        this.productRepository = productRepository;
    }

    /**
     * 多方案保费测算
     */
    public QuoteResult generateQuotes(QuoteRequest request) {
        var quotes = new ArrayList<PremiumQuote>();

        // 对每种可行的缴费期间组合生成报价方案
        var paymentPeriods = List.of("10年", "20年", "30年");
        var sumInsureds = List.of(
            new BigDecimal("100000"),
            new BigDecimal("300000"),
            new BigDecimal("500000")
        );

        for (var period : paymentPeriods) {
            for (var si : sumInsureds) {
                try {
                    var premium = premiumCalculator.calculateAnnualPremium(
                        request.productCode(),
                        request.insurant(),
                        si,
                        period,
                        request.coveragePeriod()
                    );

                    quotes.add(new PremiumQuote(
                        si, period, premium.adjustedPremium(),
                        premium.occupationMultiplier()
                    ));
                } catch (RateNotFoundException e) {
                    // 该组合没有费率，跳过
                }
            }
        }

        // 按保费从低到高排序
        quotes.sort(Comparator.comparing(PremiumQuote::annualPremium));

        // 生成方案对比说明
        var comparisonText = generateComparisonText(request, quotes);

        return new QuoteResult(quotes, comparisonText);
    }

    private String generateComparisonText(QuoteRequest request,
                                           List<PremiumQuote> quotes) {
        if (quotes.isEmpty()) {
            return "未找到符合条件的报价方案。";
        }

        var quotesText = new StringBuilder();
        for (int i = 0; i < quotes.size(); i++) {
            var q = quotes.get(i);
            quotesText.append(String.format(
                "方案%d：保额%s元，缴费%s，年保费%s元\n",
                i + 1, q.sumInsured(), q.paymentPeriod(), q.annualPremium()
            ));
        }

        return "为您生成以下报价方案：\n" + quotesText +
               "\n建议：缴费期越长，年缴保费越低但总保费越高；保额越高保障越充分。请根据预算和需求选择。";
    }
}

十一、最佳实践

11.1 AI应用开发注意事项

  1. 人工兜底：所有AI决策链路必须保留人工审核环节。核保、理赔、反欺诈的AI判断不能完全替代人工，应作为辅助决策工具。

  2. 可解释性：AI的推荐和判断必须附带理由。如核保决策必须说明依据（是哪条健康告知导致加费/拒保）。

  3. 合规性：保险行业受严格监管。AI模型不得包含歧视性变量（种族、宗教信仰等）。核保和理赔AI决策逻辑需满足监管可解释性要求。

  4. 数据安全：健康告知、医疗记录、身份证号等属于敏感个人信息（《个人信息保护法》规定）。必须加密存储、传输加密、访问审计。

  5. 幻觉防控：条款QA场景中，LLM的答案必须严格基于检索到的原文，明确标注引用来源。严禁LLM在缺乏原文依据时编造条款内容。

  6. 性能优化：

    • 向量检索使用pgvector的IVFFlat索引提升查询速度
    • 费率查询使用Redis缓存（TTL=24小时）
    • LLM调用使用Virtual Threads异步并发
    • 大文件（条款PDF）使用异步处理队列

11.2 常见问题

Q1: Spring AI 2.x 的Agent和直接调用ChatClient有什么区别？ A: ChatClient是底层API，手动管理Tool调用。Agent封装了Tool调用的编排逻辑（自动判断是否需要调用Tool、调用哪个Tool、如何将Tool结果整合到响应中），支持多步推理和复杂任务链。

Q2: 向量数据库选型建议？ A: 对于保险场景（强一致性要求高、数据量可控），推荐使用pgvector（在PostgreSQL中直接存储向量）。它的优势是与业务数据库共用同一套基础设施，支持事务，运维简单。对于超大规模（亿级向量），可考虑Milvus或Qdrant。

Q3: LLM调用延迟高怎么办？ A: (1) 使用Streaming模式让用户逐字看到输出；(2) 对常见问题缓存LLM回答；(3) 使用较小模型处理简单任务；(4) 异步非阻塞调用（Virtual Threads）。

Q4: 如何评估RAG质量？ A: 关注三个指标：(1) 召回率（检索到的相关文档数/总相关文档数）；(2) 精确率（检索到的相关文档数/检索到的总文档数）；(3) 答案忠实度（LLM答案与原文的一致程度）。建议建立人工标注的测试集定期评估。
